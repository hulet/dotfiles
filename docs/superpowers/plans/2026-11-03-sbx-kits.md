# sbx Kits in dotfiles Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the per-repo sbx kits with four composable v2 mixins in `dotfiles/kits/` (`base`, `scanners`, `cloud-clis`, `postgres-client`), referenced over `git+https`, plus a rewritten `sbxc` and a local test suite.

**Architecture:** Each kit is a directory with a `spec.yaml` (kit schema v2, `kind: mixin`); `base` also ships files into `/home/agent`. Kits combine with repeated `--kit "git+https://github.com/hulet/dotfiles.git#ref=main&dir=kits/<name>"`. `sbxc` (moved into `dotfiles/sbxc.zsh`, sourced from `zshrc`) always adds `base` and maps name arguments to that URL. `tests/sbx-kits/run.sh` checks specs, catalogs, checksum pinning, both shell scripts and `sbxc` without needing sbx.

**Tech Stack:** Docker Sandboxes (sbx) kit schema v2 YAML, POSIX sh, zsh, Python 3 + PyYAML (tests only), Claude Code settings JSON.

**Spec:** `docs/superpowers/specs/2026-11-03-sbx-kits-design.md` (dotfiles). Read it before starting.

```sh
set -u
fail=0
work="$(mktemp -d)"
trap 'rm -rf "$work"' EXIT
mkdir -p "$work/bin"
PATH="$work/bin:$PATH"

# macOS has shasum but not always sha256sum; the kit scripts call sha256sum.
if ! command -v sha256sum >/dev/null 2>&1; then
  printf '#!/bin/sh\nexec shasum -a 256 "$@"\n' > "$work/bin/sha256sum"
  chmod +x "$work/bin/sha256sum"
fi

check() { # check <description> <command...>: record a failure, keep going
  desc=$1
  shift
  if "$@"; then :; else echo "  not ok: $desc" >&2; fail=1; fi
}

has_line() { # has_line <file> <exact line>
  [ -f "$1" ] && grep -qxF -- "$2" "$1"
}

no_file() { [ ! -e "$1" ]; }

stub() { # stub <name> <sh body>: put a stub command first on PATH
  printf '#!/bin/sh\n%s\n' "$2" > "$work/bin/$1"
  chmod +x "$work/bin/$1"
}
```

- [ ] **Step 2: Write `tests/sbx-kits/check_specs.py`**

```python
"""Static checks for dotfiles/kits: fields, catalogs, checksum pinning."""
import pathlib
import re
import sys

import yaml

REQUIRED = ("schemaVersion", "kind", "name", "displayName", "description")
CATALOG_ITEM = re.compile(r"^\s*- \*\*`([a-z0-9-]+)`\*\*")
SHA256 = re.compile(r"\b[0-9a-f]{64}\b")


def catalog(text, heading):
    """Kit names listed as '- **`name`**' under a markdown heading."""
    names, inside = set(), False
    for line in text.splitlines():
        if line.strip() == heading:
            inside = True
            continue
        if inside and line.lstrip().startswith("#"):
            break
        match = CATALOG_ITEM.match(line)
        if inside and match:
            names.add(match.group(1))
    return names


def main(root):
    kits_dir = pathlib.Path(root) / "kits"
    errors = []
    names = sorted(p.name for p in kits_dir.iterdir() if p.is_dir())
    if not names:
        errors.append("no kits found under kits/")
    specs = {}
    for name in names:
        path = kits_dir / name / "spec.yaml"
        if not path.is_file():
            errors.append(f"{name}: missing spec.yaml")
            continue
        spec = yaml.safe_load(path.read_text())
        specs[name] = spec
        for field in REQUIRED:
            if not spec.get(field):
                errors.append(f"{name}: missing {field}")
        if str(spec.get("schemaVersion")) != "2":
            errors.append(f"{name}: schemaVersion must be '2'")
        if spec.get("kind") != "mixin":
            errors.append(f"{name}: kind must be mixin")
        if spec.get("name") != name:
            errors.append(f"{name}: name must equal the directory name")
        if not (spec.get("agentInstructions") or {}).get("content"):
            errors.append(f"{name}: missing agentInstructions.content")
        for i, step in enumerate((spec.get("setup") or {}).get("install") or []):
            where = f"{name}: install step {i + 1}"
            if not step.get("description"):
                errors.append(f"{where}: missing description")
            command = step.get("command", "")
            if not isinstance(command, str) or not re.search(r"\bcurl\b", command):
                continue
            if "sha256sum -c" not in command:
                errors.append(f"{where}: downloads with curl but never runs sha256sum -c")
            if len(set(SHA256.findall(command))) < 2:
                errors.append(f"{where}: needs an embedded SHA256 per architecture")
            if re.search(r"checksums|SHASUMS|\.sha256\b|\.sig\b", command):
                errors.append(f"{where}: must not fetch checksums or signatures at install time")
            if "Updating a pinned tool" not in path.read_text():
                errors.append(f"{name}: pinned steps need a comment pointing to kits/README.md")
    base = specs.get("base")
    if base:
        listed = catalog(base["agentInstructions"]["content"], "### Kit catalog")
        if listed != set(names):
            errors.append(f"base catalog lists {sorted(listed)}, kits/ has {names}")
    readme = kits_dir / "README.md"
    if readme.is_file():
        listed = catalog(readme.read_text(), "## Kit catalog")
        if listed != set(names):
            errors.append(f"kits/README.md catalog lists {sorted(listed)}, kits/ has {names}")
    else:
        errors.append("kits/README.md is missing")
    for error in errors:
        print(f"  not ok: {error}", file=sys.stderr)
    return 1 if errors else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```

```python
"""Write every curl-using install step to <out>/<kit>-<n>.sh; print the count."""
import pathlib
import re
import sys

import yaml

root, out = pathlib.Path(sys.argv[1]), pathlib.Path(sys.argv[2])
out.mkdir(parents=True, exist_ok=True)
count = 0
for spec_path in sorted((root / "kits").glob("*/spec.yaml")):
    spec = yaml.safe_load(spec_path.read_text())
    for i, step in enumerate((spec.get("setup") or {}).get("install") or []):
        command = step.get("command", "")
        if isinstance(command, str) and re.search(r"\bcurl\b", command):
            (out / f"{spec_path.parent.name}-{i + 1}.sh").write_text(command)
            count += 1
print(count)
```

```sh
#!/bin/sh
# Every install step that downloads with curl must stop before unpacking or
# running anything when the download does not match its pinned SHA256.
here="$(cd "$(dirname "$0")" && pwd)"
. "$here/lib.sh"
root=$1

count="$($PY "$here/extract_curl_steps.py" "$root" "$work/steps")"
check "found at least one download step" [ "$count" -gt 0 ]

# curl writes a tampered file to its -o target; everything that would unpack,
# install or run the download leaves a marker instead.
stub curl 'while [ $# -gt 0 ]; do [ "$1" = -o ] && { echo tampered > "$2"; shift; }; shift; done'
stub uname 'echo "$FAKE_ARCH"'
for cmd in tar unzip install ln chmod chown apt-get mv; do
  stub "$cmd" "touch \"$work/ran-$cmd\""
done

for step in "$work"/steps/*.sh; do
  for arch in x86_64 aarch64; do
    rm -f "$work"/ran-*
    if FAKE_ARCH=$arch sh "$step" >/dev/null 2>&1; then
      check "$(basename "$step") ($arch) fails on a tampered download" false
    fi
    for marker in "$work"/ran-*; do
      [ -e "$marker" ] && check "$(basename "$step") ($arch) ran ${marker##*/ran-} before verifying" false
    done
  done
done
exit $fail
```

```sh
#!/bin/sh
# Bind-mount a project's dependency and build directories from sandbox-local
# disk over the virtiofs workspace mount. Must run as root (mount).
#
# Why: the workspace is the host's checkout. Directories holding
# platform-specific builds would otherwise be shared, so the host's macOS
# install and the sandbox's Linux install overwrite each other; and some build
# tools lose races on virtiofs itself (rolldown's parallel mkdir fails with
# "File exists (os error 17)" in .svelte-kit). Each mount is private to the
# sandbox; the host keeps using its own directory.
#
# The list is detected from files in the workspace root (the table below),
# so a repo without those files gets no mounts. Arguments after the
# workspace, or SCRATCH_MOUNT_DIRS (space-separated), replace detection.
# Idempotent: an existing mount is skipped. Never creates a missing parent
# (e.g. node_modules/.cache when node_modules is absent). Covers the
# workspace root only, not git worktrees under it.
#
# The workspace comes from $1, falling back to $WORKSPACE_DIR: sudo strips the
# variable, both in the manual re-apply command and in the SessionStart hook
# that works around docker/sbx-releases#420 (see kits/base/spec.yaml).
set -eu
workspace="${1:-${WORKSPACE_DIR:-}}"
[ -n "$workspace" ] || exit 0
[ "$#" -eq 0 ] || shift
scratch_root="${SCRATCH_MOUNT_ROOT:-/var/tmp/sbx-workspace}"

has() { # has <file-or-glob...>: does the workspace root contain any of them?
  for pattern in "$@"; do
    for path in "$workspace"/$pattern; do
      [ -e "$path" ] && return 0
    done
  done
  return 1
}

if [ "$#" -gt 0 ]; then
  dirs="$*"
elif [ -n "${SCRATCH_MOUNT_DIRS:-}" ]; then
  dirs="$SCRATCH_MOUNT_DIRS"
else
  # Detector table: project has X -> make Y sandbox-private. Add a line per
  # ecosystem (e.g. Cargo.toml -> target, Gemfile -> vendor/bundle).
  dirs=""
  if has package.json; then dirs="$dirs node_modules"; fi
  if has pyproject.toml 'requirements*.txt'; then dirs="$dirs .venv"; fi
  if has svelte.config.js svelte.config.ts svelte.config.mjs; then dirs="$dirs .svelte-kit"; fi
fi

repo_name="$(basename "$workspace")"
for rel in $dirs; do
  dir="$workspace/$rel"
  [ -d "$(dirname "$dir")" ] || continue
  mountpoint -q "$dir" && continue
  scratch="$scratch_root/$repo_name-$(echo "$rel" | tr '/' '-')"
  mkdir -p "$scratch" "$dir"
  chown 1000:1000 "$scratch"
  mount --bind "$scratch" "$dir"
  echo "sbx-workspace-mounts: mounted $scratch on $dir"
done
```

```sh
# Lockfile installs, stamps, the mount-point safety check and failure
# handling of sbx-deps-install.sh.

. "$here/lib.sh"

echo node_modules > "$work/mounted"
: > "$work/log"

sh "$script" "$r" >/dev/null 2>&1
check "first run: npm ci" has_line "$work/log" "npm ci"
sh "$script" "$r" >/dev/null 2>&1
check "unchanged lockfile: no second install" [ "$(wc -l < "$work/log")" -eq 1 ]
echo v2 > "$r/package-lock.json"
sh "$script" "$r" >/dev/null 2>&1
check "changed lockfile: installs again" [ "$(wc -l < "$work/log")" -eq 2 ]

echo v3 > "$r/package-lock.json"
: > "$work/log"
NPM_EXIT=1 sh "$script" "$r" >/dev/null 2>&1
status=$?
check "failed install still exits 0" [ "$status" -eq 0 ]
sh "$script" "$r" >/dev/null 2>&1
check "failed install is retried next run" [ "$(grep -c '^npm ci$' "$work/log")" -eq 2 ]

r="$work/unmounted"
mkdir -p "$r/node_modules"
echo v1 > "$r/package-lock.json"
: > "$work/mounted"
: > "$work/log"
out="$(sh "$script" "$r" 2>&1)"
check "never installs into a directory that is not a mount" [ ! -s "$work/log" ]

r="$work/py"

echo .venv > "$work/mounted"
: > "$work/log"
sh "$script" "$r" >/dev/null 2>&1
check "uv.lock: uv sync --frozen" has_line "$work/log" "uv sync --frozen"

r="$work/none"
mkdir -p "$r"
: > "$work/log"
```

```sh
# Install a project's dependencies from its lockfile into the sandbox-private
# directories that sbx-workspace-mounts.sh mounts. Runs as the agent user, at
# sandbox start and (docker/sbx-releases#420 workaround) at every session
# start.
#
# Only an unambiguous lockfile triggers an install: package-lock.json ->
# `npm ci`, uv.lock -> `uv sync --frozen`. Other projects (pip -e ., yarn,
# pnpm) are left to their own instructions. An install runs only when the
# lockfile's hash differs from the stamp left by the last successful install,
# so an unchanged project costs nothing.
#
# It installs only into a directory that is a mount point, so it can never
# write into the host's own node_modules or .venv if the mount is missing.
# Failures warn and exit 0: the sandbox should start even when an install
# cannot; the next start retries.
set -u
workspace="${1:-${WORKSPACE_DIR:-}}"
[ -n "$workspace" ] || exit 0
cd "$workspace" || exit 0

install_from() { # install_from <lockfile> <private-dir> <command...>
  lockfile=$1
  dir=$2
  shift 2
  [ -f "$lockfile" ] || return 0
  if ! mountpoint -q "$dir"; then
    echo "sbx-deps-install: $dir is not sandbox-private; not installing from $lockfile" >&2
    return 0
  fi
  stamp="$dir/.sbx-deps-install.sha256"
  hash="$(sha256sum "$lockfile" | cut -d' ' -f1)"
  [ "$(cat "$stamp" 2>/dev/null)" = "$hash" ] && return 0
  echo "sbx-deps-install: $lockfile changed; running $*"
  if "$@"; then
    echo "$hash" > "$stamp"
  else
    echo "sbx-deps-install: '$*' failed; the next start retries" >&2
  fi
}

install_from package-lock.json node_modules npm ci
install_from uv.lock .venv uv sync --frozen
exit 0
```

````markdown
# sbx kits

Composable [Docker Sandboxes](https://docs.docker.com/ai/sandboxes/) kits
(schema v2 mixins) for Claude Code sandboxes. Every sandbox gets `base`; add

## Kit catalog

- **`base`** — Node.js 24, sandbox-private `node_modules` / `.venv` /
  `.svelte-kit` with lockfile installs, the superpowers plugin, the Claude Code
  settings file, and the general package and registry hosts. In every sandbox.

## Using the kits

Each kit is referenced by URL; the repo is public, so no credentials are
needed:

```

```

From a project directory, on the host:

```sh
K='git+https://github.com/hulet/dotfiles.git#ref=main&dir=kits'
sbx run claude . --kit "$K/base" --kit "$K/scanners" \
  -- --settings /home/agent/sbx-claude-settings.json

Or use `sbxc` from dotfiles, which always adds `base`:

sbxc scanners postgres-client   # create, or re-attach if it exists
sbxc                            # re-attach
```

  `sbx run`, re-attaches included; it is a Claude Code argument sbx does not

  work in progress, `sbx rm <sandbox>`, and create it again.

4. Try it in a throwaway sandbox (below), then push to `main`.

## Updating a pinned tool

Every binary downloaded from a URL is pinned to a version and to a SHA256 per
architecture embedded in its install step, checked with `sha256sum -c` before
anything is unpacked. Checksums are never fetched at install time from the
same release as the binary: whoever controls a release controls its checksum
file, and in 2026 Trivy's release tags were re-pointed to a malicious build.
apt (signed indexes) and PyPI via uv (immutable release files) need only a
version pin.

To update one:

1. **Pick the version.** Prefer a release at least 7 days old: supply-chain
   compromises are usually caught within days. Read the release notes for
   breaking changes and security advisories.
2. **Get the expected hashes** for that exact version, for both
   `linux/amd64` and `linux/arm64`, from the vendor's published source:
   Trivy's `trivy_<version>_checksums.txt`, gitleaks'
   `gitleaks_<version>_checksums.txt`, hadolint's GitHub asset digests
   (`gh api repos/hadolint/hadolint/releases/tags/v<version>`), Node's
   `SHASUMS256.txt`, the gcloud archive's SHA256 from the download, the AWS
   CLI zip's SHA256 after its signature check (step 3).
3. **Cross-check through a second channel**; if anything disagrees, stop and
   report it — do not update:
   - Trivy: `cosign verify-blob <asset> --bundle <asset>.sigstore.json --certificate-identity-regexp 'https://github.com/aquasecurity/trivy/.github/workflows/.+' --certificate-oidc-issuer https://token.actions.githubusercontent.com`
   - Node: `gpg --verify SHASUMS256.txt.asc` with the keys from
     https://github.com/nodejs/release-keys imported.
   - AWS CLI: `gpg --verify <zip>.sig <zip>` against the AWS CLI Team key
     (fingerprint `FB5D B77F D5C1 18B8 0511 ADA8 A631 0ACC 4672 475C`, from
     AWS's install documentation).
   - hadolint: compare with `docker run --rm --entrypoint sha256sum hadolint/hadolint:v<version>-debian /bin/hadolint`
     (that image's architecture).
   - gitleaks: compare the release's GitHub asset digests with its
     checksums file.
4. **Edit the kit:** in the one install step, change `version=` and both
   `sum=` values together. Nothing else.
5. **Test:** run `tests/sbx-kits/run.sh`; then create a throwaway sandbox with
   the changed kit by local path (above) and check that the install step
   passed, `<tool> --version` reports the new version, and the tool's normal
   command works.
6. **Commit** one tool per commit; the message records the old and new
   version, where the hashes came from and how they were cross-checked.

- In-sandbox names use the `sbx-` prefix (`~/.local/lib/sbx-*`,
  `/opt/sbx-<kit>/`).
- Run `tests/sbx-kits/run.sh` before every push that touches `kits/` or
````

```zsh
# sbxc [kit ...] [-- claude-args]: create or re-attach to this directory's
# Claude Code sandbox. Every sandbox gets the base kit from dotfiles/kits;
# each name argument adds kits/<name> (kits/README.md lists them), so a new
# kit needs no change here. Kits, like --no-share-skills (opting out of the
# shared skills store), apply only at creation, so a re-attach passes neither.
# --settings is a claude argument sbx does not remember; it is passed every
# time.

# The sandbox sbx creates for this directory when no --name is given.
_sbxc_sandbox_name() {
    local name="claude-$(basename "$PWD" | tr -cs 'A-Za-z0-9.+-' '-')"
    print -r -- "${name%-}"
}

sbxc() {
    emulate -L zsh
    local kit_url='git+https://github.com/hulet/dotfiles.git#ref=main&dir=kits'
    local -a kits=(base) kit_flags=()
    local -a claude_args=(--dangerously-skip-permissions --settings /home/agent/sbx-claude-settings.json)
    local kit name

    while (( $# )) && [[ $1 != -* ]]; do
        # Names only, never paths or URLs: kits come from dotfiles alone.
        if [[ ! $1 =~ '^[a-z0-9-]+$' ]]; then
            print -u2 "sbxc: '$1' is not a kit name (see ~/git/dotfiles/kits/README.md)"
            return 1
        fi
        kits+=("$1")
        shift
    done
    [[ ${1-} == -- ]] && shift

    name="$(_sbxc_sandbox_name)"
    if sbx ls -q 2>/dev/null | grep -qxF -- "$name"; then
        if (( ${#kits} > 1 )); then
            print -u2 "sbxc: $name already exists and kits are fixed at creation."
            print -u2 "sbxc: to change them, push your work, then: sbx rm $name && sbxc ${kits[2,-1]}"
        fi
        sbx run --name "$name" -- "${claude_args[@]}" "$@"
    else
        for kit in ${(u)kits}; do
            kit_flags+=(--kit "$kit_url/$kit")
        done
        sbx run claude . --no-share-skills "${kit_flags[@]}" -- "${claude_args[@]}" "$@"
    fi
}
```
